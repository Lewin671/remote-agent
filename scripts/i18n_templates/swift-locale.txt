
public enum MessageLocale {
    public static let appleTags: [String: String] = [APPLE_TAGS]
    private static let languages: [String: String] = [LANGUAGE_MAP]
    public static func resolve(_ preference: String = "auto", system: [String] = Locale.preferredLanguages) -> String {
        if appleTags[preference] != nil { return preference }
        for tag in system {
            let base = tag.replacingOccurrences(of: "_", with: "-").lowercased()
                .components(separatedBy: ".")[0].components(separatedBy: "@")[0]
                .components(separatedBy: "-")[0]
            if let language = languages[base] { return language }
        }
        return "zh-CN"
    }
}
