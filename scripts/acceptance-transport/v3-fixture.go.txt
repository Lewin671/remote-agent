// Compile only against the pristine generation-3 checkout for offline acceptance.
package main
import (
 "encoding/json"
 "os"
 "path/filepath"
 "strings"
 "github.com/Lewin671/remote-agent/internal/deviceauth"
)
func main() {
 home:=os.Getenv("REMOTE_AGENT_HOME")
 if !strings.HasPrefix(home,"/tmp/") && !strings.HasPrefix(home,"/private/tmp/") { panic("isolated home required") }
 var d struct {
  Pin deviceauth.Pin
  States []deviceauth.State
  Proofs []*deviceauth.RecoveryStatement
  ID string
  EdSeed,XPrivate []byte
  Lease deviceauth.LeaseCertificate
 }
 if err:=json.NewDecoder(os.Stdin).Decode(&d);err!=nil{panic("invalid fixture")}
 ep,err:=deviceauth.InitEndpoint(filepath.Join(home,"security-v3.db"),d.Pin,d.States,d.Proofs,d.ID,d.EdSeed,d.XPrivate,d.Lease)
 if err!=nil{panic("fixture init failed")}
 if err:=ep.Close();err!=nil{panic("fixture close failed")}
}
